/// 获取标准玻璃设置
static LiquidGlassSettings getStandardSettings({
  double? blur,
  double? thickness,
}) {
  final glassColor = globalUseDarkMode
      ? Colors.black.withValues(alpha: 0.4)  // 从 0.35 增加到 0.4
      : Colors.white.withValues(alpha: 0.08);
  
  return LiquidGlassSettings(
    glassColor: glassColor,
    blur: blur ?? 15.0,
    thickness: thickness ?? 1.0,
  );
}

/// 获取卡片玻璃设置
static LiquidGlassSettings getCardSettings({
  double? alpha,
}) {
  final glassColor = globalUseDarkMode
      ? Colors.black.withValues(alpha: alpha ?? 0.35)  // 从 0.3 增加到 0.35
      : Colors.white.withValues(alpha: alpha ?? 0.05);
  
  return LiquidGlassSettings(
    glassColor: glassColor,
    blur: 12.0,
    thickness: 0.8,
  );
}

/// 获取按钮玻璃设置
static LiquidGlassSettings getButtonSettings({
  bool isSelected = false,
  Color? selectedColor,
}) {
  Color glassColor;
  
  if (isSelected) {
    final baseColor = selectedColor ?? Colors.blue;
    glassColor = globalUseDarkMode
        ? baseColor.withValues(alpha: 0.45)  // 从 0.4 增加到 0.45
        : baseColor.withValues(alpha: 0.3);
  } else {
    glassColor = globalUseDarkMode
        ? Colors.white.withValues(alpha: 0.1)  // 从 0.08 增加到 0.1
        : Colors.white.withValues(alpha: 0.05);
  }
  
  return LiquidGlassSettings(
    glassColor: glassColor,
    blur: 0,
    thickness: 10,
  );
}

/// 获取对话框玻璃设置
static LiquidGlassSettings getDialogSettings() {
  final glassColor = globalUseDarkMode
      ? Colors.black.withValues(alpha: 0.5)  // 从 0.45 增加到 0.5
      : Colors.white.withValues(alpha: 0.15);
  
  return LiquidGlassSettings(
    glassColor: glassColor,
    blur: 25.0,
    thickness: 18.0,
  );
}

static LiquidGlassSettings getInputSettings() {

### 4. 设置页面动画优化

**需求：**
- 设置界面进入二级/三级页面时容易掉帧卡顿
- 动画过快不流畅
- 切换时降低上一页渲染质量或分辨率

**建议实现方案：**

1. **使用 RepaintBoundary 隔离重绘**
```dart
RepaintBoundary(
  child: SettingsGeneralScreen(),
)
```

2. **页面切换时降低玻璃质量**
```dart
class SettingsPageTransition extends PageTransitionsBuilder {
  @override
  Widget buildTransitions<T>(
    PageRoute<T> route,
    BuildContext context,
    Animation<double> animation,
    Animation<double> secondaryAnimation,
    Widget child,
  ) {
    // 在动画进行时降低上一页的玻璃质量
    return AnimatedBuilder(
      animation: secondaryAnimation,
      builder: (context, child) {
        final isExiting = secondaryAnimation.value > 0;
        return GlassQualityOverride(
          quality: isExiting ? GlassQuality.low : GlassQuality.standard,
          child: child!,
        );
      },
      child: child,
    );
  }
}
```

3. **使用截图替代实时渲染**
```dart
// 在页面切换开始时截图
final screenshot = await _captureWidget(context);

// 在动画期间显示截图
Stack(
  children: [
    if (isAnimating) Image.memory(screenshot),
    if (!isAnimating) ActualPage(),
  ],
)
```
