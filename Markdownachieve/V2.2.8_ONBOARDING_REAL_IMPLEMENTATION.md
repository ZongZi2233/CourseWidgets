```dart
enum ThemeMode {
  defaultTheme,    // 默认主题
  materialYou,     // Material You 系统颜色
  monet,           // 莫奈取色
  custom,          // 自定义颜色

class _OnboardingThemeState extends State<OnboardingTheme> {
  ThemeMode _themeMode = ThemeMode.defaultTheme;
  Color? _customColor;
  
  // 主题模式选择
  Widget _buildThemeModeSelector() {
    return Column(
      children: [
        _buildThemeModeOption(
          '默认主题',
          '使用应用默认的粉色主题',
          ThemeMode.defaultTheme,
        ),
        _buildThemeModeOption(
          'Material You',
          '跟随系统动态颜色',
          ThemeMode.materialYou,
        ),
        _buildThemeModeOption(
          '莫奈取色',
          '从背景图片提取主色调',
          ThemeMode.monet,
        ),
        _buildThemeModeOption(
          '自定义颜色',
          '选择你喜欢的颜色',
          ThemeMode.custom,
        ),
      ],
    );
  }
  
  // 如果选择自定义，显示颜色选择器
  Widget _buildColorPicker() {
    if (_themeMode != ThemeMode.custom) return SizedBox();
    
    return Wrap(
      spacing: 12,
      runSpacing: 12,
      children: _themeColors.map((color) {
        return GestureDetector(
          onTap: () => setState(() => _customColor = color),
          child: Container(
            width: 60,
            height: 60,
            decoration: BoxDecoration(
              color: color,
              borderRadius: BorderRadius.circular(16),
              border: _customColor == color
                  ? Border.all(color: Colors.white, width: 3)
                  : null,
            ),
          ),
        );
      }).toList(),
    );
  }
}
```

```dart
class ThemeService {
  static final ThemeService _instance = ThemeService._internal();
  factory ThemeService() => _instance;
  ThemeService._internal();
  
  final StorageService _storage = StorageService();
  
  // 保存主题模式
  Future<void> saveThemeMode(String mode) async {
    await _storage.setString('theme_mode', mode);
  }
  
  // 获取主题模式
  String getThemeMode() {
    return _storage.getString('theme_mode') ?? 'default';
```

## 📝 代码示例

### 完整的主题页面实现

```dart
class _OnboardingThemeState extends State<OnboardingTheme> {
  ThemeMode _themeMode = ThemeMode.defaultTheme;
  Color? _customColor;
  
  @override
  Widget build(BuildContext context) {
    return Scaffold(
      body: Column(
        children: [
          // Header with glass card
          SafeArea(
            child: Padding(
              padding: const EdgeInsets.all(20),
              child: liquid.LiquidCard(
                borderRadius: 24,
                padding: 20,
                glassColor: Colors.white.withValues(alpha: 0.05),
                child: Column(
                  children: [
                    Text('个性化', style: ...),
                    Text('选择你喜欢的主题', style: ...),
                  ],
                ),
              ),
            ),
          ),
          
          // Content
          Expanded(
            child: ListView(
              children: [
                // 主题模式选择
                _buildThemeModeCard(),
                
                // 如果选择自定义，显示颜色选择器
                if (_themeMode == ThemeMode.custom)
                  _buildColorPickerCard(),
                
                // 背景图片
                _buildBackgroundCard(),
              ],
            ),
          ),
          
          // Navigation
          SafeArea(
            child: Padding(
              padding: const EdgeInsets.all(20),
              child: Column(
                children: [
                  liquid.LiquidButton(
                    text: '进入课程表',
                    onTap: _completeOnboarding,
                  ),
                  liquid.LiquidButton(
                    text: '上一步',
                    onTap: widget.onBack,
                  ),
                ],
              ),
            ),
          ),
        ],
      ),
    );
  }
  
  Future<void> _completeOnboarding() async {
    // 保存主题设置
    final themeService = ThemeService();
    await themeService.saveThemeMode(_themeMode.toString());
    
    if (_themeMode == ThemeMode.custom && _customColor != null) {
      await themeService.saveCustomColor(_customColor!);
    }
    
    // 完成引导
    widget.onComplete();
  }
}
```
